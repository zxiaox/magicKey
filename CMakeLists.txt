cmake_minimum_required(VERSION 3.20)
project(MagicKey VERSION 1.5.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()
configure_file(cmake/app_config.hpp.in generated/app_config.hpp @ONLY)
configure_file(src/app.manifest.in generated/app.manifest @ONLY)
configure_file(THIRD-PARTY-NOTICES.txt generated/THIRD-PARTY-NOTICES.txt COPYONLY)
configure_file(assets/MagicKey.ico generated/MagicKey.ico COPYONLY)
configure_file(assets/warcraft-frozen-throne.jpg generated/warcraft-frozen-throne.jpg COPYONLY)
if(WIN32)
  enable_language(RC)
  add_executable(MagicKey WIN32 src/main.cpp src/updater.cpp src/theme_ui.cpp src/game_notifications.cpp src/app_identity.cpp src/app.rc)
  target_include_directories(MagicKey PRIVATE src third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(MagicKey PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(MagicKey PRIVATE comctl32 user32 gdi32 winhttp bcrypt shell32 dwmapi uxtheme gdiplus shlwapi propsys ole32)
  if(MSVC)
    target_link_options(MagicKey PRIVATE /MANIFEST:NO)
  endif()
  add_executable(update_test_launcher tests/update_launcher.cpp src/updater.cpp)
  target_include_directories(update_test_launcher PRIVATE third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(update_test_launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(update_test_launcher PRIVATE user32 winhttp bcrypt shell32)
  add_executable(game_notification_tests tests/game_notification_tests.cpp src/game_notifications.cpp)
  set_target_properties(game_notification_tests PROPERTIES OUTPUT_NAME War3)
  target_compile_definitions(game_notification_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(game_notification_tests PRIVATE user32)
  add_executable(game_window_detection_tests tests/game_notification_tests.cpp src/game_notifications.cpp)
  target_compile_definitions(game_window_detection_tests PRIVATE MAGICKEY_WINDOW_FALLBACK_TEST UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(game_window_detection_tests PRIVATE user32)
  add_executable(update_test_payload tests/update_payload.cpp)
  target_compile_definitions(update_test_payload PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MINGW)
    target_link_options(MagicKey PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(update_test_launcher PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(update_test_payload PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME core COMMAND core_tests)
add_executable(update_tests tests/update_tests.cpp)
target_include_directories(update_tests PRIVATE third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_test(NAME updates COMMAND update_tests)
add_executable(theme_tests tests/theme_tests.cpp)
add_test(NAME themes COMMAND theme_tests)

if(WIN32)
  add_test(NAME game_notifications COMMAND game_notification_tests)
  add_test(NAME game_window_detection COMMAND game_window_detection_tests)
  set_tests_properties(game_notifications game_window_detection PROPERTIES TIMEOUT 30)
endif()
